```python
@tool(tier=0)
def list_inbox(count: int = 10) -> str:
    """List recent inbox emails with subject, sender, date, and read status.

    Args:
        count: Number of recent emails to list (default 10, max 25).
    """
    app = _get_outlook()
    if app is None:
        return _DEGRADED_MSG
    count = min(count, 25)

    def _fetch():
        pythoncom.CoInitialize()  # D-03: idempotent, thread-safe
        ns = app.GetNamespace("MAPI")
        inbox = ns.GetDefaultFolder(OL_FOLDER_INBOX)
        items = inbox.Items
        items.Sort("[ReceivedTime]", True)  # descending
        lines = []
        for i, item in enumerate(items):
            if i >= count:
                break
            try:
                status = "unread" if item.UnRead else "read"
                lines.append(
                    f"{i+1}. [{status}] {item.Subject}\n"
                    f"   From: {item.SenderName}\n"
                    f"   Date: {item.ReceivedTime.strftime('%Y-%m-%d %H:%M')}\n"
                    f"   ID: {item.EntryID}"
                )
            except Exception:
                continue
        return "\n".join(lines) if lines else "Inbox is empty."

    return _com_retry(_fetch)
```

```python
def read_email(entry_id: str) -> str:
    """Read the full content of an email by its ID.

    Args:
        entry_id: The EntryID of the email to read (from list_inbox results).
    """
    app = _get_outlook()
    if app is None:
        return _DEGRADED_MSG

    def _fetch():
        pythoncom.CoInitialize()  # D-03: idempotent, thread-safe
        ns = app.GetNamespace("MAPI")
        try:
            item = ns.GetItemFromID(entry_id)
        except Exception:
            return "[error] Email not found. It may have been moved or deleted."
        body = item.Body or ""
        total_len = len(body)
        if total_len > 4096:
            body = body[:4096] + f"\n[...truncated at 4KB -- {total_len // 1024}KB total]"
        mark_external_content()  # D-11: SAFE-04 quarantine
        return (
            f"Subject: {item.Subject}\n"
            f"From: {item.SenderName}\n"
            f"Date: {item.ReceivedTime.strftime('%Y-%m-%d %H:%M')}\n"
            f"To: {item.To}\n\n"
            f"{body}"
        )

    return _com_retry(_fetch)
```

```python
@tool(tier=1)
def draft_email(to: str, subject: str, body: str) -> str:
    """Draft an email and save it to the Drafts folder. Does NOT send.

    Args:
        to: Recipient email address(es), comma-separated for multiple.
        subject: Email subject line.
        body: Email body text.
    """
    app = _get_outlook()
    if app is None:
        return _DEGRADED_MSG

    def _create():
        pythoncom.CoInitialize()  # D-03: idempotent, thread-safe
        mail = app.CreateItem(OL_MAIL_ITEM)
        mail.To = to
        mail.Subject = subject
        mail.Body = body
        mail.Save()  # Saves to Drafts -- does NOT send (D-05)
        return f"Draft saved. draft_id: {mail.EntryID}  Subject: '{subject}'  To: {to}"

    return _com_retry(_create)
```

2. **send_draft** (tier=2 per D-04 -- destructive/irreversible, gets dry-run + quarantine check automatically from call_tool):
```python
@tool(tier=2)
def send_draft(draft_id: str) -> str:
    """Send a previously drafted email. Requires draft_id from draft_email.

    Args:
        draft_id: The EntryID of the draft to send (from draft_email result).
    """
    app = _get_outlook()
    if app is None:
        return _DEGRADED_MSG

    def _send():
        pythoncom.CoInitialize()  # D-03: idempotent, thread-safe
        ns = app.GetNamespace("MAPI")
        try:
            draft = ns.GetItemFromID(draft_id)
        except Exception:
            return "[error] Draft not found. It may have been deleted or moved from Drafts."
        subj = draft.Subject
        to = draft.To
        draft.Send()
        return f"Email sent. Subject: '{subj}' To: {to}"

    return _com_retry(_send)
```

Key details per decisions:
- draft_email is tier=1 (D-05): reversible, user can delete from Drafts
- send_draft is tier=2 (D-04): call_tool automatically writes dry-run to actions.log and checks SAFE-04 quarantine before executing
- Two-turn speed bump (D-06): send_draft requires draft_id which only comes from a prior draft_email result. The model must relay the draft_id to the user in turn N, then the user says "send it" in turn N+1
- draft_email calls .Save() NEVER .Send()
- send_draft captures Subject/To before .Send() because the item becomes invalid after sending
- Both _create() and _send() closures call pythoncom.CoInitialize() as first line per D-03
