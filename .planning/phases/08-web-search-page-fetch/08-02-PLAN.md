```python
"""fetch_page -- Tier 0 read-only tool. Extracts page content via trafilatura.

SAFE-07: Only fetches URLs returned by search_web in the current turn.
SAFE-04: Marks the turn as having external content (quarantine).
"""
import requests as http_requests
import trafilatura

from jarvis.tools._registry import tool, mark_external_content
from jarvis.tools.search import is_url_allowed

_FETCH_TIMEOUT = 12  # seconds (D-04 recommendation: 10-15s)
_TEXT_CAP = 4096     # 4 KB cap matching read_file/read_pdf (D-04)
_POOR_THRESHOLD = 50 # words below this = "poor" extraction quality


@tool(tier=0)
def fetch_page(url: str) -> str:
    """Fetch and extract the main text content of a web page. Only accepts URLs returned by search_web in this turn.

    Args:
        url: The URL to fetch (must be from current search results).
    """
    # SAFE-07: URL allow-list check (D-01, D-02)
    if not is_url_allowed(url):
        return (
            "URL not in current search results. fetch_page only accepts URLs "
            "returned by search_web in this turn. Ask me to search first, "
            "then I can fetch a specific result."
        )

    # Fetch with requests (not trafilatura.fetch_url) for URL control (D-03)
    try:
        resp = http_requests.get(
            url,
            timeout=_FETCH_TIMEOUT,
            allow_redirects=True,
            headers={"User-Agent": "Mozilla/5.0 (compatible; Jarvis/1.0)"},
        )
        resp.raise_for_status()
    except Exception as exc:
        return f"[error] Failed to fetch {url}: {exc}"

    final_url = resp.url
    html = resp.text

    # Extract content via trafilatura (bare_extraction for metadata access)
    doc = trafilatura.bare_extraction(
        html,
        url=final_url,
        favor_precision=True,
        include_tables=True,
        include_comments=False,
    )

    title = "(no title)"
    text = ""
    if doc is not None:
        title = doc.get("title", "") or "(no title)" if isinstance(doc, dict) else getattr(doc, "title", "") or "(no title)"
        text = doc.get("text", "") or "" if isinstance(doc, dict) else getattr(doc, "text", "") or ""

    word_count = len(text.split()) if text else 0
    quality = "clean" if word_count >= _POOR_THRESHOLD else "poor"

    # 4 KB text cap (D-04)
    total_len = len(text)
    truncation = ""
    if total_len > _TEXT_CAP:
        text = text[:_TEXT_CAP]
        truncation = f"[...truncated at 4KB -- {total_len // 1024}KB total]"

    # SAFE-04: Mark external content — blocks Tier 2 tools for rest of turn (D-06)
    mark_external_content()

    # Build response with metadata (D-05)
    parts = [
        f"Title: {title}",
        f"Source: {final_url}",
        f"Extraction: {quality}",
        f"Words: {word_count}",
    ]
    if truncation:
        parts.append(truncation)
    parts.append("")
    parts.append(text)
    return "\n".join(parts)
```

Note on `bare_extraction` return type: trafilatura 2.x may return either a dict or a Document object depending on version. The code handles both with the `isinstance(doc, dict)` check plus `getattr` fallback. This is defensive coding against API variance.

class SearXNGComponent(_ComponentBase):
    """Pings SearXNG at startup. Yellow if unreachable, green if reachable."""

    def __init__(self, url: str = "http://localhost:8888") -> None:
        super().__init__("Web")
        self._url = url

    def try_init(self) -> bool:
        try:
            import requests as http_requests
            resp = http_requests.get(
                f"{self._url}/search",
                params={"q": "test", "format": "json"},
                timeout=3,

            resp.raise_for_status()
            return self._succeed()
        except Exception as exc:
            # Degraded, not failed — SearXNG is optional (D-13)
            self._status = "degraded"
            self._error = str(exc)

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| fetch_page -> internet | Outbound HTTP to arbitrary web servers (constrained by URL allow-list) |
| web content -> LLM context | Untrusted HTML content extracted and passed to model |
| SearXNG results -> URL set | Search engine results define the set of fetchable URLs |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-04 | Elevation | fetch_page SSRF | mitigate | `is_url_allowed(url)` check before any HTTP request; only SearXNG-returned URLs permitted (SAFE-07) |
| T-08-05 | Tampering | web page content injection | mitigate | `mark_external_content()` triggers SAFE-04 quarantine blocking Tier 2 tools; result wrapped in `<untrusted_content>` by assistant.py (SAFE-06) |
| T-08-06 | Information Disclosure | page content leaks to model | accept | By design: user requests page content; 4KB cap limits exposure; model context is ephemeral |
| T-08-07 | Denial of Service | slow/large page fetch | mitigate | 12-second timeout on requests.get; 4KB text cap on extracted content |
| T-08-08 | Spoofing | model fabricates URL | mitigate | `_current_turn_urls` populated only by search_web; cleared every turn; fetch_page refuses unknown URLs |
| T-08-09 | Tampering | stale URLs from previous turn | mitigate | `clear_turn_urls()` called at turn start in _turn_pipeline alongside quarantine reset |
</threat_model>
